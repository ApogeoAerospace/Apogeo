39 const bool has_id = json.contains("id");
40 if (!has_request_id && !has_id) {
41 if (error) {
42 *error = "Missing required 'request_id' or 'id' field";
43 }
44 return false;
45 }
46

48 if (!request_id_json.is_string() || request_id_json.get<std::string>().empty()) {
49 if (error) {
50 *error = "Field 'request_id' or 'id' must be a non-empty string";
51 }
52 return false;
53 }
54
55 out.request_id = request_id_json.get<std::string>();
56 out.payload = json.value("payload", nlohmann::json::object());
