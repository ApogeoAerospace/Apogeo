41void plugin_log(int32_t level, const char* component, const std::string& message) {
42 HostLogFn host_log = g_host_log_fn.load(std::memory_order_acquire);
43 if (host_log != nullptr) {
44 void* user_data = g_host_user_data.load(std::memory_order_relaxed);
45 host_log(level, component, message.c_str(), user_data);
46 return;
47 }

49 // Fallback for isolated plugin tests without host.
50 std::cout << "[" << (component ? component : "Plugin") << "] " << message << std::endl;

57/**
58 * @brief Receives optional host services (includes shared logger).
59 */
60PLUGIN_EXPORT void plugin_set_host_services(const PluginHostServices* services) {
61 if (services != nullptr && services->log != nullptr) {
62 // If api_version mismatches, fallback to local plugin logging behavior.
63 if (services->api_version != kSupportedHostServicesApiVersion) {
64 g_host_log_fn.store(nullptr, std::memory_order_release);
65 g_host_user_data.store(nullptr, std::memory_order_relaxed);
66
67 bool expected = false;
68 if (g_host_api_mismatch_warned.compare_exchange_strong(expected, true, std::memory_order_acq_rel)) {
69 std::cout << "[Structures] WARNING: Host services api_version mismatch (host="
70 << services->api_version
71 << ", plugin-supported="
72 << kSupportedHostServicesApiVersion
73 << "). Falling back to local plugin logging." << std::endl;
74 }
75 return;
76 }
77
78 g_host_api_mismatch_warned.store(false, std::memory_order_release);
79 g_host_user_data.store(services->user_data, std::memory_order_relaxed);
80 g_host_log_fn.store(services->log, std::memory_order_release);
81 return;
82 }
83
84 g_host_log_fn.store(nullptr, std::memory_order_release);
85 g_host_user_data.store(nullptr, std::memory_order_relaxed);
86}

89 * @brief Creates Structures plugin instance and loads initial placeholders.
90 */
91PLUGIN_EXPORT PluginHandle plugin_create_instance() {
92 auto* instance = new StructuresPluginInstance();
93
94 const bool mass_loaded = instance->module.loadMassPropertiesFromJson(kDefaultMassPropsPath);
95 const bool limits_loaded = instance->module.loadStructuralLimitsFromCsv(kDefaultStructuralLimitsPath);
96 if (!mass_loaded || !limits_loaded) {
97 plugin_log(PLUGIN_LOG_ERROR, "Structures", "Failed to initialize default structural sources."
98 " mass_loaded=" + std::to_string(mass_loaded)
99 + " limits_loaded=" + std::to_string(limits_loaded));
100 delete instance;
101 return nullptr;
102 }
103
104 instance->initialized = true;
105
106 return reinterpret_cast<PluginHandle>(instance);
107}

109/**
110 * @brief Configures structural data paths from general configuration. (No actuators)
111 */
112PLUGIN_EXPORT int32_t plugin_configure(PluginHandle handle, const char* json_params) {
113 if (!handle || !json_params) {
114 return -1;
115 }
116
117 auto* instance = reinterpret_cast<StructuresPluginInstance*>(handle);
118 if (!instance->initialized) {
119 return -2;
120 }
121
122 try {
123 const nlohmann::json params = nlohmann::json::parse(json_params);
124
125 std::string mass_json_path = kDefaultMassPropsPath;
126 std::string limits_csv_path = kDefaultStructuralLimitsPath;
127
128 if (params.contains("mass_properties_path")) {
129 mass_json_path = params["mass_properties_path"].get<std::string>();
130 }
131 if (params.contains("structural_limits_path")) {
132 limits_csv_path = params["structural_limits_path"].get<std::string>();
133 }
134 if (params.contains("debug_output")) {
135 instance->debug_output = params["debug_output"].get<bool>();
136 }
137
138 // Mass/CoM/inertia are temporarily kept in JSON placeholders.
139 const bool mass_loaded = instance->module.loadMassPropertiesFromJson(mass_json_path);
140 const bool limits_loaded = instance->module.loadStructuralLimitsFromCsv(limits_csv_path);
141
142 if (!mass_loaded || !limits_loaded) {
143 if (!mass_loaded) {
144 plugin_log(PLUGIN_LOG_ERROR, "Structures", "Failed to open JSON file: " + mass_json_path);
145 }
146 if (!limits_loaded) {
147 plugin_log(PLUGIN_LOG_ERROR, "Structures", "Failed to open CSV file: " + limits_csv_path);
148 }
149 return -4;
150 }
151
152 return 0;
153 } catch (...) {
154 return -3;
155 }
156}

158/**
159 * @brief Executes a type-1 structural tick (state read-only, force/torque output).
160 *
161 * @warning This implementation intentionally does not apply active physics.
162 * It returns neutral output (zeros) to preserve flow compatibility.
163 */
164PLUGIN_EXPORT int32_t plugin_tick(PluginHandle handle, PluginTickData* data) {
165 if (!handle || !data || !data->state_buffer) {
166 return -1;
167 }
168
169 PluginVector3* force_output = data->output_force ? data->output_force : data->force_out;
170 PluginVector3* torque_output = data->output_torque ? data->output_torque : data->torque_out;
171 if (!force_output) {
172 return -1;
173 }
174
175 auto* instance = reinterpret_cast<StructuresPluginInstance*>(handle);
176 if (!instance->initialized) {
177 return -2;
178 }
179
180 const auto* state = state_vector::GetGeneralState(data->state_buffer);
181 if (!state || !state->position() || !state->velocity()) {
182 return -3;
183 }
184
185 force_output->x = 0.0f;
186 force_output->y = 0.0f;
187 force_output->z = 0.0f;
188
189 if (torque_output) {
190 torque_output->x = 0.0f;
191 torque_output->y = 0.0f;
192 torque_output->z = 0.0f;
193 }
194
195 return 0;
196}
