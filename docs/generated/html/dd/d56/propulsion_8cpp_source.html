75// Funciones auxiliares
76double calculateAtmosphericPressure(double altitude) {
77 // Modelo de atmósfera estándar
78 if (altitude < 0) altitude = 0;
79
80 if (altitude <= 11000.0) {
81 // Troposfera
82 double temperature = 288.15 - 0.0065 * altitude;
83 return ATMOSPHERIC_PRESSURE_SL * pow(temperature / 288.15, 5.256);
84 } else {
85 // Estratosfera simplificada
86 return ATMOSPHERIC_PRESSURE_SL * exp(-altitude / 8400.0);
87 }
88}

90double calculateThrustAtAltitude(double vacuum_thrust, double sea_level_thrust,
91 double nozzle_exit_area, double ambient_pressure) {
92 // Ecuación de empuje con compensación de altitud
93 // T = T_vac - (P_ambient - P_exit) * A_exit
94 // Simplificación: asumimos P_exit = 0 para toberas optimizadas para vacío
95 double pressure_thrust_loss = ambient_pressure * nozzle_exit_area;
96 double altitude_thrust = vacuum_thrust - pressure_thrust_loss;
97
98 // Limitar entre empuje a nivel del mar y empuje en vacío
99 return std::max(sea_level_thrust, std::min(vacuum_thrust, altitude_thrust));
100}

102double calculateSpecificImpulseAtAltitude(double isp_vacuum, double isp_sea_level,
103 double ambient_pressure) {
104 // Interpolación lineal basada en presión atmosférica
105 double pressure_ratio = ambient_pressure / ATMOSPHERIC_PRESSURE_SL;
106 return isp_sea_level + (isp_vacuum - isp_sea_level) * (1.0 - pressure_ratio);
107}

109double calculateMassFlowRate(double thrust, double specific_impulse) {
110 // Ecuación de Tsiolkovsky: T = Isp * g0 * dm/dt
111 if (specific_impulse <= 0) return 0.0;
112 return thrust / (specific_impulse * STANDARD_GRAVITY);
113}

271PLUGIN_EXPORT int32_t plugin_tick(PluginHandle handle, PluginTickData* data) {
272 if (!handle || !data || !data->state_buffer) {
273 return -1; // Error: parámetros inválidos
274 }
275
276 // Compatibilidad con ambas versiones de la API
277 PluginVector3* force_output = data->output_force ? data->output_force : data->force_out;
278 PluginVector3* torque_output = data->output_torque ? data->output_torque : data->torque_out;
279
280 if (!force_output) {
281 return -1; // Error: sin puntero de salida para fuerzas
282 }
283
284 PropulsionPluginInstance* instance = reinterpret_cast<PropulsionPluginInstance*>(handle);
285 if (!instance->initialized) {
286 return -2; // Error: plugin no inicializado
287 }
288
289 // Obtener estado actual del FlatBuffer
290 auto state = state_vector::GetGeneralState(data->state_buffer);
291 if (!state) return -3;
292
293 // Extraer datos del estado
294 double pos_x = state->position()->x();
295 double pos_y = state->position()->y();
296 double pos_z = state->position()->z();
297 double vel_x = state->velocity()->x();
298 double vel_y = state->velocity()->y();
299 double vel_z = state->velocity()->z();
300
301 // Calcular altitud (WGS84 consistent with core)
302 const double EARTH_RADIUS = 6378137.0;
303 double altitude = sqrt(pos_x*pos_x + pos_y*pos_y + pos_z*pos_z) - EARTH_RADIUS;
304 if (altitude < 0) altitude = 0;
305
306 // Actualizar estado interno
307 instance->current_altitude = altitude;
308 instance->current_atmospheric_pressure = calculateAtmosphericPressure(altitude);
309
310 // Verificar si hay combustible
311 if (instance->enable_fuel_consumption && instance->current_fuel_mass <= 0.0) {
312 instance->engine_on = false;
313 instance->current_fuel_mass = 0.0;
314 }
315
316 // Inicializar fuerzas
317 double force_x = 0.0, force_y = 0.0, force_z = 0.0;
318
319 // Calcular empuje si el motor está encendido
320 if (instance->engine_on && instance->current_fuel_mass > 0.0) {
321 // Actualizar tiempo de quemado
322 instance->burn_time += data->delta_time;
323
324 // Calcular empuje en función de la altitud
325 double base_thrust = instance->enable_altitude_compensation ?
326 calculateThrustAtAltitude(instance->vacuum_thrust, instance->sea_level_thrust,
327 instance->nozzle_exit_area, instance->current_atmospheric_pressure) :
328 instance->sea_level_thrust;
329
330 // Aplicar throttling si está habilitado
331 if (instance->enable_throttling) {
332 base_thrust *= instance->throttle_setting;
333 }
334
335 // Aplicar eficiencias
336 base_thrust *= instance->combustion_efficiency * instance->nozzle_efficiency;
337
338 instance->current_thrust = base_thrust;
339
340 // Calcular consumo de combustible
341 if (instance->enable_fuel_consumption) {
342 double current_isp = instance->enable_altitude_compensation ?
343 calculateSpecificImpulseAtAltitude(instance->specific_impulse_vac,
344 instance->specific_impulse_sl,
345 instance->current_atmospheric_pressure) :
346 instance->specific_impulse_sl;
347
348 instance->current_mass_flow_rate = calculateMassFlowRate(base_thrust, current_isp);
349
350 // Consumir combustible
351 double fuel_consumed = instance->current_mass_flow_rate * data->delta_time;
352 instance->current_fuel_mass = std::max(0.0, instance->current_fuel_mass - fuel_consumed);
353 }
354
355 // Calcular dirección del empuje
356 // Simplificación: empuje en dirección de la velocidad (o hacia arriba si velocidad es cero)
357 double velocity_magnitude = sqrt(vel_x*vel_x + vel_y*vel_y + vel_z*vel_z);
358
359 double thrust_dir_x, thrust_dir_y, thrust_dir_z;
360 if (velocity_magnitude > 0.1) {
361 // Empuje en dirección de la velocidad
362 thrust_dir_x = vel_x / velocity_magnitude;
363 thrust_dir_y = vel_y / velocity_magnitude;
364 thrust_dir_z = vel_z / velocity_magnitude;
365 } else {
366 // Empuje hacia arriba si no hay velocidad significativa
367 thrust_dir_x = 0.0;
368 thrust_dir_y = 0.0;
369 thrust_dir_z = 1.0;
370 }
371
372 // Aplicar vectorización de empuje si está habilitada
373 if (instance->enable_thrust_vectoring && instance->thrust_vector_angle != 0.0) {
374 // Simplificación: rotación en el plano XZ
375 double cos_angle = cos(instance->thrust_vector_angle);
376 double sin_angle = sin(instance->thrust_vector_angle);
377
378 double new_thrust_x = thrust_dir_x * cos_angle - thrust_dir_z * sin_angle;
379 double new_thrust_z = thrust_dir_x * sin_angle + thrust_dir_z * cos_angle;
380
381 thrust_dir_x = new_thrust_x;
382 thrust_dir_z = new_thrust_z;
383 }
384
385 // Aplicar empuje
386 force_x = base_thrust * thrust_dir_x;
387 force_y = base_thrust * thrust_dir_y;
388 force_z = base_thrust * thrust_dir_z;
389
390 } else {
391 instance->current_thrust = 0.0;
392 instance->current_mass_flow_rate = 0.0;
393 }
394
395 // Asignar fuerzas de salida
396 force_output->x = static_cast<float>(force_x);
397 force_output->y = static_cast<float>(force_y);
398 force_output->z = static_cast<float>(force_z);
399
400 // Torque (para thrust vectoring)
401 if (torque_output) {
402 torque_output->x = 0.0;
403 torque_output->y = 0.0;
404 torque_output->z = 0.0;
405
406 if (instance->enable_thrust_vectoring && instance->thrust_vector_angle != 0.0) {
407 // Torque proporcional al ángulo de vectorización
408 torque_output->y = static_cast<float>(instance->current_thrust *
409 sin(instance->thrust_vector_angle) * 2.0);
410 }
411 }
412
413 return 0; // Éxito
414}
