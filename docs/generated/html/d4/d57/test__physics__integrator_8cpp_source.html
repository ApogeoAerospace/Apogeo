259 PhysicsIntegrator integ(type);

262

293TEST_F(PhysicsIntegratorTest, TorqueFreeSymmetricBodySpinStability) {
294 PhysicsState s;
295 s.mass = 50.0;
296 s.inertia = makeInertiaTensor(10.0, 10.0, 5.0, 0.0, 0.0, 0.0);
297 s.angular_velocity = Vector3(0.0, 0.0, 10.0);
298
299 Vector3 force = Vector3::Zero();
300 Vector3 torque = Vector3::Zero();
301 double dt = 0.001;
302
303 integrator->setIntegratorType(PhysicsIntegrator::IntegratorType::RUNGE_KUTTA_4);
304
305 double initial_energy = 0.5 * 5.0 * 10.0 * 10.0; // 0.5 * Izz * wz²

323// ---------------------------------------------------------------------------
324// Test: Free precession of an asymmetric rigid body (energy conservation)
325// ---------------------------------------------------------------------------
326// Body with inertia Ixx=10, Iyy=20, Izz=30 and omega_0=(1, 2, 3) rad/s.
327// Without external torque, rotational kinetic energy T and angular momentum
328 // L² = (I*omega)·(I*omega) should be conserved.
329TEST_F(PhysicsIntegratorTest, TorqueFreeAsymmetricBodyEnergyConservation) {
330 PhysicsState s;
331 s.mass = 100.0;
332 s.inertia = makeInertiaTensor(10.0, 20.0, 30.0, 0.0, 0.0, 0.0);
333 s.angular_velocity = Vector3(1.0, 2.0, 3.0);
334
335 Vector3 force = Vector3::Zero();
336 Vector3 torque = Vector3::Zero();
337 double dt = 0.0005;
338
339 integrator->setIntegratorType(PhysicsIntegrator::IntegratorType::RUNGE_KUTTA_4);
340
341 // Compute initial invariants
342 Vector3 L0 = s.inertia * s.angular_velocity;
343 double T0 = 0.5 * s.angular_velocity.dot(L0);
344 double L0_sq = L0.squaredNorm();
345
346 // Integrate 5 s (10000 steps)
347 for (int i = 0; i < 10000; ++i) {
348 s = integrator->integrate(s, force, torque, dt);
349 }
350
351 // Compute final invariants
352 Vector3 Lf = s.inertia * s.angular_velocity;
353 double Tf = 0.5 * s.angular_velocity.dot(Lf);
354 double Lf_sq = Lf.squaredNorm();
355
356 // Rotational kinetic energy conserved (tolerance ~0.1%)
357 EXPECT_NEAR(Tf, T0, T0 * 1e-3);
358
359 // Angular-momentum magnitude conserved (tolerance ~0.1%)
360 EXPECT_NEAR(Lf_sq, L0_sq, L0_sq * 1e-3);
361
362 // Quaternion remains unitary
363 EXPECT_NEAR(s.orientation.norm(), 1.0, 1e-10);
364}
