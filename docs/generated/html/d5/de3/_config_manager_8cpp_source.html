19bool ConfigManager::loadConfig(const std::string& config_file) {
20 try {
21 std::ifstream file(config_file);
22 if (!file.is_open()) {
23 // Use default values.
24 setDefaults();
25 config_loaded_ = false;
26 logLoadFailure("Config file not found, using defaults: " + config_file);
27 return false; // Defaults are applied, but no file was loaded
28 }
29
30 nlohmann::json config_json;
31 file >> config_json;
32 file.close();
33
34 std::string error_detail;
35
36 // Parse main sections
37 if (!parseSimulationConfig(config_json, error_detail)) {
38 setDefaults();
39 config_loaded_ = false;
40 logLoadFailure("Failed to parse simulation config: " + error_detail + ". Using defaults");
41 return false;
42 }
43 if (!parsePluginConfigs(config_json, error_detail)) {
44 setDefaults();
45 config_loaded_ = false;
46 logLoadFailure("Failed to parse plugin config: " + error_detail + ". Using defaults");
47 return false;
48 }
49
50 // File paths
51 if (config_json.contains("initial_state_file")) {
52 initial_state_file_ = config_json["initial_state_file"];
53 } else {
54 initial_state_file_ = "data/default_state.json";
55 }
56
57 if (config_json.contains("output_directory")) {
58 output_directory_ = config_json["output_directory"];
59 } else {
60 output_directory_ = "output/";
61 }
62
63 // Reset before parsing so a missing section in a new config clears a previous value.
64 flight_script_path = "";
65 if (config_json.contains("vehicle_models") && config_json["vehicle_models"].contains("flight_computer")) {
66 flight_script_path = config_json["vehicle_models"]["flight_computer"]["script_path"];
67 }
68
69 // Validate overall consistency
70 if (!validateConfig(error_detail)) {
71 setDefaults();
72 config_loaded_ = false;
73 logLoadFailure("Configuration validation failed: " + error_detail + ". Using defaults");
74 return false;
75 }
76
77 config_loaded_ = true;
78 // Intentionally no runtime side effects here.
79 // Logger bootstrap (console/file/level) is owned by main.cpp so
80 // ConfigManager stays focused on load/parse/validate responsibilities.
81 return true;
82
83 } catch (const std::exception& e) {
84 // Read/parse failure: use defaults as fallback
85 setDefaults();
86 config_loaded_ = false;
87 logLoadFailure("Error loading config: " + std::string(e.what()));
88 return false;
89 }
90}

92bool ConfigManager::saveConfig(const std::string& config_file) const {
93 try {

99 config_json["simulation"]["enable_logging"] = simulation_config_.enable_logging;
100 config_json["simulation"]["log_file"] = simulation_config_.log_file;
101 config_json["simulation"]["log_level"] = simulation_config_.log_level;
102 config_json["ipc"]["tick_event_interval"] = simulation_config_.ipc_tick_event_interval;
103 config_json["ipc"]["telemetry_interval_ticks"] = simulation_config_.ipc_telemetry_interval_ticks;
104 config_json["logging"]["console_output"] = simulation_config_.console_output;
105 config_json["logging"]["file_output"] = simulation_config_.file_output;
106
107 // Plugin configuration
108 for (const auto& plugin : plugin_configs_) {
109 nlohmann::json plugin_json;
110 plugin_json["name"] = plugin.name;
111 plugin_json["type"] = plugin.type;
112 plugin_json["library_path"] = plugin.library_path;
113 plugin_json["enabled"] = plugin.enabled;
114 plugin_json["parameters"] = plugin.parameters;
115 config_json["plugins"].push_back(plugin_json);
116 }
117
118 // File paths
119 config_json["initial_state_file"] = initial_state_file_;
120 config_json["output_directory"] = output_directory_;
121
122 std::ofstream file(config_file);
123 if (!file.is_open()) {
124 LOG_ERROR("Cannot open config file for writing: " + config_file, "ConfigManager");
125 return false;
126 }
127
128 file << config_json.dump(4);
129 file.close();
130
131 LOG_INFO("Configuration saved to: " + config_file, "ConfigManager");
132 return true;
133
134 } catch (const std::exception& e) {
135 LOG_ERROR("Error saving config: " + std::string(e.what()), "ConfigManager");
136 return false;

164 try {

168 simulation_config_.max_iterations = sim.value("max_iterations", 10000);
169 simulation_config_.enable_logging = sim.value("enable_logging", true);
170 simulation_config_.log_file = sim.value("log_file", "logs/molab.log");
171 simulation_config_.log_level = sim.value("log_level", "INFO");
172 } else {
173 // If section is missing, use defaults
174 simulation_config_.simulation_duration = 100.0;
175 simulation_config_.max_iterations = 10000;
176 simulation_config_.enable_logging = true;
177 simulation_config_.log_file = "logs/molab.log";
178 simulation_config_.log_level = "INFO";
179 }
180
181 if (json.contains("logging") && json["logging"].is_object()) {
182 simulation_config_.console_output = json["logging"].value("console_output", true);

184 } else {
185 simulation_config_.console_output = true;

204bool ConfigManager::parsePluginConfigs(const nlohmann::json& json, std::string& error_detail) {
205 try {
206 plugin_configs_.clear();
207
208 if (json.contains("plugins") && json["plugins"].is_array()) {
209 for (const auto& plugin_json : json["plugins"]) {
210 PluginConfig plugin;
211 plugin.name = plugin_json.value("name", "");
212 plugin.type = plugin_json.value("type", 0);
213 plugin.library_path = plugin_json.value("library_path", "");
214 plugin.enabled = plugin_json.value("enabled", true);
215
216 if (plugin_json.contains("parameters")) {
217 plugin.parameters = plugin_json["parameters"];
218 }
219
220 // Add only if name and valid path are present
221 if (!plugin.name.empty() && !plugin.library_path.empty()) {
222 plugin_configs_.push_back(plugin);
223 }
224 }
225 }
226
227 return true;
228 } catch (const std::exception& e) {
229 error_detail = e.what();
230 return false;

234bool ConfigManager::validateConfig(std::string& error_detail) const {
235 // Basic simulation validation
236 if (simulation_config_.simulation_duration <= 0) {
237 error_detail = "simulation.duration must be positive (got " + std::to_string(simulation_config_.simulation_duration) + ")";
238 return false;
239 }
240
241 if (simulation_config_.max_iterations <= 0) {
242 error_detail = "simulation.max_iterations must be positive (got " + std::to_string(simulation_config_.max_iterations) + ")";
243 return false;
244 }
245
246 if (simulation_config_.ipc_tick_event_interval <= 0) {
247 error_detail = "ipc.tick_event_interval must be positive (got " + std::to_string(simulation_config_.ipc_tick_event_interval) + ")";
248 return false;
249 }
250
251 if (simulation_config_.ipc_telemetry_interval_ticks <= 0) {
252 error_detail = "ipc.telemetry_interval_ticks must be positive (got " + std::to_string(simulation_config_.ipc_telemetry_interval_ticks) + ")";
253 return false;
254 }
255
256 // Plugin validation
257 for (const auto& plugin : plugin_configs_) {
258 if (plugin.name.empty()) {
259 error_detail = "plugin name cannot be empty";
260 return false;
261 }
262
263 if (plugin.library_path.empty()) {
264 error_detail = "plugin library_path cannot be empty";
265 return false;
266 }
267
268 if (plugin.type < 0 || plugin.type > 1) {
269 error_detail = "plugin type must be 0 or 1 (got " + std::to_string(plugin.type) + ")";
270 return false;
271 }
272 }
273
274 return true;
275}

277void ConfigManager::logLoadFailure(const std::string& message) {
278 auto& logger = Logger::getInstance();
279
280 // On configuration load failure, force full logger visibility so diagnostics
281 // are not lost even if console output was disabled in previous runtime state.
282 logger.setConsoleOutputEnabled(true);
283 logger.setLogLevel(LogLevel::DEBUG);
284 if (!simulation_config_.log_file.empty()) {
285 logger.setLogFile(simulation_config_.log_file);
286 }
287
288 logger.error(message, "ConfigManager");
289}
