20inline bool hasValidMass(double mass) {
21 return std::isfinite(mass) && mass > kMassEpsilon;

24inline Quaternion4 normalizedOrIdentity(const Quaternion4& q) {
25 if (!q.coeffs().allFinite()) {
26 return Quaternion4::Identity();
27 }
28
29 const double norm = q.norm();
30 if (norm < kQuatNormEpsilon) {
31 return Quaternion4::Identity();
32 }
33
34 Quaternion4 normalized = q;
35 normalized.normalize();
36 return normalized;
37}

69// ---------------------------------------------------------------------------
70// computeOdeDerivatives — ODE system core
71// ---------------------------------------------------------------------------
72// Operates directly on OdeState, extracting only temporary Eigen vectors
73// required for linear algebra operations (cross product,
74// quaternion product, LDLT solve). Avoids rebuilding a full
75// PhysicsState (including Matrix3d inertia copy) on each evaluation.
76
77void PhysicsIntegrator::computeOdeDerivatives(const OdeState& y, OdeState& dydt,
78 double mass,
79 const InertiaTensor3& inertia,
80 const Vector3& force,
81 const Vector3& torque,
82 const Vector3& gravity_acceleration) {
83 dydt.fill(0.0);
84
85 // Extract dynamic variables from flat vector
86 const Vector3 vel = odeVelocity(y);
87 const Vector3 omega = odeAngularVelocity(y);
88 const Quaternion4 q = normalizedOrIdentity(odeOrientation(y));
89
90 // --- Linear kinematics: x_dot = v ---
91 dydt[0] = vel.x();
92 dydt[1] = vel.y();
93 dydt[2] = vel.z();
94
95 // --- Newton's second law: v_dot = F / m + g ---
96 Vector3 linear_acceleration = gravity_acceleration;
97 if (hasValidMass(mass)) {
98 linear_acceleration += (force / mass);
99 }
100
101 if (linear_acceleration.allFinite()) {
102 dydt[3] = linear_acceleration.x();
103 dydt[4] = linear_acceleration.y();
104 dydt[5] = linear_acceleration.z();
105 } else {
106 dydt[3] = 0.0; dydt[4] = 0.0; dydt[5] = 0.0;
107 }
108
109 // --- Rotational kinematics: q_dot = 0.5 * q * (0, omega) ---
110 // Hamilton product between orientation quaternion and pure
111 // angular-velocity quaternion, scaled by 0.5.
112 Quaternion4 omega_quat(0.0, omega.x(), omega.y(), omega.z());
113 Quaternion4 q_dot;
114 q_dot.coeffs() = (q * omega_quat).coeffs() * 0.5;
115 dydt[6] = q_dot.w();
116 dydt[7] = q_dot.x();
117 dydt[8] = q_dot.y();
118 dydt[9] = q_dot.z();
119
120 // --- Ecuaciones de Euler rotacionales: ---
121 // omega_dot = I^(-1) * (tau - omega x (I * omega))
122 // Uses LDLT decomposition (stable for symmetric
123 // positive semidefinite matrices) instead of explicit inversion.
124 Vector3 i_omega = inertia * omega;
125 Vector3 gyroscopic = omega.cross(i_omega);
126 Vector3 net_torque = torque - gyroscopic;
127
128 Vector3 alpha = Vector3::Zero();
129 if (inertia.allFinite() && net_torque.allFinite()) {
130 Eigen::LDLT<InertiaTensor3> ldlt(inertia);
131 if (ldlt.info() == Eigen::Success) {
132 alpha = ldlt.solve(net_torque);
133 if (ldlt.info() != Eigen::Success || !alpha.allFinite()) {
134 alpha = Vector3::Zero();
135 }
136 }
137 }
138 dydt[10] = alpha.x();
139 dydt[11] = alpha.y();
140 dydt[12] = alpha.z();
141}

147PhysicsState PhysicsIntegrator::integrate(const PhysicsState& state,
148 const Vector3& force,
149 const Vector3& torque,
150 double dt,
151 const Vector3& gravity_acceleration) {
152 if (!std::isfinite(dt)) {
153 LOG_WARNING("Invalid time step (NaN/Inf), skipping integration step", "PhysicsIntegrator");
154 return state;
155 }
156 OdeState y = state.toOdeState();
157
158 // ODE system lambda. Captures constant step parameters
159 // (mass, inertia, forces) by reference without copying.
160 auto system = [&](const OdeState& y_in, OdeState& dydt, double /*t*/) {
161 computeOdeDerivatives(y_in, dydt, state.mass, state.inertia, force, torque, gravity_acceleration);
162 };
163
164 switch (integrator_type_) {
165 case IntegratorType::EULER: {
166 odeint::euler<OdeState> stepper;
167 stepper.do_step(system, y, state.time, dt);
168 break;
169 }
170 case IntegratorType::RUNGE_KUTTA_4: {
171 odeint::runge_kutta4<OdeState> stepper;
172 stepper.do_step(system, y, state.time, dt);
173 break;
174 }
175 case IntegratorType::VERLET: {
176 // Velocity-Verlet for translation + RK4 for rotation.
177 // Odeint does not provide velocity_verlet for mixed
178 // first- and second-order systems, so this is manual.
179
180 // --- Translation: Velocity-Verlet (a = F/m + g) ---
181 Vector3 a_n = gravity_acceleration;
182 if (hasValidMass(state.mass)) {
183 a_n += (force / state.mass);
184 }
185
186 Vector3 v_half = state.velocity + a_n * (0.5 * dt);
187 Vector3 new_pos = state.position + v_half * dt;
188
189 // Constant acceleration during step => a_{n+1} = a_n
190 Vector3 new_vel = v_half + a_n * (0.5 * dt);
191
192 // --- Rotation: explicit RK4 over (q, omega) ---
193 OdeState y_rot = state.toOdeState();
194 odeint::runge_kutta4<OdeState> rot_stepper;
195 rot_stepper.do_step(system, y_rot, state.time, dt);
196
197 // Apply Verlet translation + RK4 rotation
198 odePackVec3(y, 0, new_pos);
199 odePackVec3(y, 3, new_vel);
200 y[6] = y_rot[6];
201 y[7] = y_rot[7];
202 y[8] = y_rot[8];
203 y[9] = y_rot[9];
204 y[10] = y_rot[10];
205 y[11] = y_rot[11];
206 y[12] = y_rot[12];
207 break;
208 }
209 default: {
210 LOG_WARNING("Unknown integrator type, falling back to Runge-Kutta 4", "PhysicsIntegrator");
211 odeint::runge_kutta4<OdeState> stepper;
212 stepper.do_step(system, y, state.time, dt);
213 break;
214 }
215 }
216
217 // Unpack result and normalize quaternion
218 PhysicsState new_state = state; // preserves mass and inertia
219 new_state.fromOdeState(y);
220 new_state.time = state.time + dt;
221 new_state.orientation = normalizedOrIdentity(new_state.orientation);
222
223 return new_state;
224}

226void PhysicsIntegrator::setIntegratorType(const std::string& type_name) {
227 if (type_name == "euler") {
228 integrator_type_ = IntegratorType::EULER;
229 } else if (type_name == "runge_kutta_4") {
230 integrator_type_ = IntegratorType::RUNGE_KUTTA_4;
231 } else if (type_name == "verlet") {
232 integrator_type_ = IntegratorType::VERLET;
233 } else {
234 LOG_WARNING("Unknown integrator type: " + type_name + ", using Runge-Kutta 4", "PhysicsIntegrator");
235 integrator_type_ = IntegratorType::RUNGE_KUTTA_4;
236 }
237
238 LOG_INFO("Integrator type changed to: " + getIntegratorTypeName(), "PhysicsIntegrator");
239}

250PhysicsState PhysicsIntegrator::fromFlatBuffer(const state_vector::GeneralState* fb_state) {
251// Converts serialized state (FlatBuffer) to internal PhysicsState (double).
252 // float->double conversion happens here; reverse truncation is
253 // PluginManager responsibility when rebuilding the buffer.
254
255 if (fb_state == nullptr) {
256 LOG_WARNING("Null FlatBuffer state pointer, returning default PhysicsState", "PhysicsIntegrator");
257 return PhysicsState{};
258 }
259 PhysicsState state;
260
261 if (fb_state->position()) {
262 state.position = Vector3(fb_state->position()->x(),
263 fb_state->position()->y(),
264 fb_state->position()->z());
265 }
266
267 if (fb_state->velocity()) {
268 state.velocity = Vector3(fb_state->velocity()->x(),
269 fb_state->velocity()->y(),
270 fb_state->velocity()->z());
271 }
272
273 if (fb_state->orientation()) {
274 // Eigen::Quaterniond constructor: (w, x, y, z)
275 state.orientation = Quaternion4(
276 fb_state->orientation()->w(),
277 fb_state->orientation()->x(),
278 fb_state->orientation()->y(),
279 fb_state->orientation()->z()
280 );
281 }
282
283 if (fb_state->angular_velocity()) {
284 state.angular_velocity = Vector3(fb_state->angular_velocity()->x(),
285 fb_state->angular_velocity()->y(),
286 fb_state->angular_velocity()->z());
287 }
288
289 if (fb_state->inertia_tensor()) {
290 state.inertia = makeInertiaTensor(
291 fb_state->inertia_tensor()->ixx(), fb_state->inertia_tensor()->iyy(),
292 fb_state->inertia_tensor()->izz(), fb_state->inertia_tensor()->ixy(),
293 fb_state->inertia_tensor()->ixz(), fb_state->inertia_tensor()->iyz());
294 }
295
296 state.orientation = normalizedOrIdentity(state.orientation);
297 state.mass = std::isfinite(fb_state->total_mass()) ? std::max(0.0, static_cast<double>(fb_state->total_mass())) : 0.0;
298 state.time = std::isfinite(fb_state->sim_time()) ? static_cast<double>(fb_state->sim_time()) : 0.0;
299 return state;
300}
