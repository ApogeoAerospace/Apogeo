48void FlightComputerContext::register_api() {
49 // ==========================================
50 // C++ TO LUA API BRIDGE
51 // ==========================================
52
53 // READ FUNCTIONS: Allow Lua to read simulation telemetry
54 lua_.set_function("get_altitude", [&]() -> float {
55 return current_altitude_;
56 });
57
58 lua_.set_function("get_vertical_velocity", [&]() -> float {
59 return current_vertical_velocity_;
60 });
61
62 // WRITE FUNCTIONS -> Allow Lua to send commands to the simulation
63 lua_.set_function("set_throttle", [&](float value) {
64 // Clamp throttle between 0% (0.0) and 100% (1.0)
65 throttle_cmd_ = std::clamp(value, 0.0f, 1.0f);
66 });
67
68 // UTILITY FUNCTIONS -> Allow Lua to print to the simulation console
69 lua_.set_function("print_log", [](std::string msg) {
70 LOG_INFO(msg, "FlightComputer");
71 });

77 if (!in) return;
78
79 // ==========================================
80 // STEP 1: READ SIMULATION STATE (TELEMETRY)
81 // ==========================================
82 // Read current state from the physics engine into internal variables for Lua access.
83 if (in->position()) {
84 current_altitude_ = in->position()->z();
85 } else {
86 current_altitude_ = 0.0f;
87 }
88
89 if (in->velocity()) {
90 current_vertical_velocity_ = in->velocity()->z();
91 } else {
92 current_vertical_velocity_ = 0.0f;

95 // Reset commands to zero before executing the script
96 throttle_cmd_ = 0.0f;
97
98 // ==========================================
99 // STEP 2: EXECUTE USER SCRIPT
100 // ==========================================
101 if (valid_) {
102 // Layer 2: Runtime protection - Catch script errors gracefully
103 sol::protected_function on_tick = lua_["on_tick"];
104 if (on_tick.valid()) {
105 auto result = on_tick(dt);
106 if (!result.valid()) {
107 sol::error err = result;
108 LOG_ERROR("Error executing on_tick: " + std::string(err.what()), "FlightComputer");
109 }
110 } else {
111 LOG_ERROR("Function 'on_tick' not found in script", "FlightComputer");
112 }
113 }
114
115 // ==========================================
116 // STEP 3: REBUILD STATE (SEND COMMANDS)
117 // ==========================================
118 // FlatBuffers are immutable, so we must copy the entire state and insert our new commands.
119 auto position = state_vector::Vec3(
120 in->position() ? in->position()->x() : 0.0f,
121 in->position() ? in->position()->y() : 0.0f,
122 in->position() ? in->position()->z() : 0.0f);

129 in->orientation() ? in->orientation()->y() : 0.0f,
130 in->orientation() ? in->orientation()->z() : 0.0f,
131 in->orientation() ? in->orientation()->w() : 1.0f);
132 auto angular_velocity = state_vector::Vec3(
133 in->angular_velocity() ? in->angular_velocity()->x() : 0.0f,
134 in->angular_velocity() ? in->angular_velocity()->y() : 0.0f,
135 in->angular_velocity() ? in->angular_velocity()->z() : 0.0f);

141 in->cg_location() ? in->cg_location()->z() : 0.0f);

166 flatbuffers::Offset<flatbuffers::Vector<const state_vector::EngineCmd*>> engines_fb;
167 if (auto engines = in->engines()) {

170 for (size_t i = 0; i < engines->size(); ++i) {

172 const state_vector::Vec3 tvc = ec->tvc_angles();
173 // The script command overrides throttle on engine[0].
174 // A PARALLEL_PHYSICS_CALCULATOR propulsion plugin is expected
175 // to read this value and produce the corresponding thrust force.
176 float throttle = (i == 0) ? throttle_cmd_ : ec->throttle();
177 eng_vec.emplace_back(throttle, tvc);
178 }
179 engines_fb = fbb.CreateVectorOfStructs(eng_vec);

208 if (propellant_masses_fb.o != 0) gs_builder.add_propellant_masses(propellant_masses_fb);
