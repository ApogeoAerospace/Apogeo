49static bool wait_for_file_non_empty(const std::string& path, int attempts = 50, int delay_ms = 10) {
50 for (int i = 0; i < attempts; ++i) {
51 std::ifstream file(path);
52 if (file.is_open()) {
53 file.seekg(0, std::ios::end);
54 if (file.tellg() > 0) {
55 return true;
56 }
57 }
58 std::this_thread::sleep_for(std::chrono::milliseconds(delay_ms));
59 }
60 return false;
61}
