106double calculateComponentMass(const StructuralComponent& comp) {
107 double volume;
108
109 if (comp.thickness > 0) {
110 // Estructura hueca (tubo)
111 double outer_radius = comp.diameter / 2.0;
112 double inner_radius = outer_radius - comp.thickness;
113 volume = M_PI * comp.length * (outer_radius*outer_radius - inner_radius*inner_radius);
114 } else {
115 // Estructura sólida (cilindro)
116 double radius = comp.diameter / 2.0;
117 volume = M_PI * radius * radius * comp.length;
118 }
119
120 double material_density;
121 switch (comp.material) {
122 case ALUMINUM: material_density = ALUMINUM_DENSITY; break;
123 case STEEL: material_density = STEEL_DENSITY; break;
124 case CARBON_FIBER: material_density = CARBON_FIBER_DENSITY; break;
125 default: material_density = ALUMINUM_DENSITY; break;
126 }
127
128 return volume * material_density;
129}

131double calculateMomentOfInertia(const StructuralComponent& comp, char axis) {
132 double mass = comp.mass;
133 double length = comp.length;
134 double radius = comp.diameter / 2.0;
135
136 // Momentos de inercia para cilindro
137 switch (axis) {
138 case 'x': // Longitudinal
139 return 0.5 * mass * radius * radius;
140 case 'y': // Transversal
141 case 'z':
142 return mass * (3.0 * radius * radius + length * length) / 12.0;
143 default:
144 return 0.0;
145 }
146}

148double calculateNaturalFrequency(double young_modulus, double moment_of_inertia,
149 double mass, double length) {
150 // Frecuencia natural de viga en voladizo (aproximación)
151 // f = (λ²/2π) * sqrt(EI / (mL⁴))
152 // donde λ ≈ 1.875 para el primer modo
153 double lambda = 1.875;
154 return (lambda * lambda / (2.0 * M_PI)) *
155 sqrt((young_modulus * moment_of_inertia) / (mass * pow(length, 4)));
156}

158double calculateStructuralStress(double force, double cross_sectional_area) {
159 return force / cross_sectional_area;
160}

261PLUGIN_EXPORT int32_t plugin_tick(PluginHandle handle, PluginTickData* data) {
262 if (!handle || !data || !data->state_buffer) {
263 return -1; // Error: parámetros inválidos
264 }
265
266 StructuresPluginInstance* instance = reinterpret_cast<StructuresPluginInstance*>(handle);
267 if (!instance->initialized) {
268 return -2; // Error: plugin no inicializado
269 }
270
271 // Obtener estado actual del FlatBuffer
272 auto state = state_vector::GetGeneralState(data->state_buffer);
273 if (!state) return -3;
274
275 // Extraer datos del estado
276 double pos_x = state->position()->x();
277 double pos_y = state->position()->y();
278 double pos_z = state->position()->z();
279 double vel_x = state->velocity()->x();
280 double vel_y = state->velocity()->y();
281 double vel_z = state->velocity()->z();
282
283 // Calcular aceleración (aproximación por diferencias finitas)
284 double velocity_magnitude = sqrt(vel_x*vel_x + vel_y*vel_y + vel_z*vel_z);
285 double acceleration = std::abs(velocity_magnitude - instance->previous_acceleration) / data->delta_time;
286 instance->previous_acceleration = velocity_magnitude;
287
288 // Actualizar máxima aceleración
289 instance->max_acceleration = std::max(instance->max_acceleration, acceleration);
290
291 // Calcular presión dinámica
292 double altitude = sqrt(pos_x*pos_x + pos_y*pos_y + pos_z*pos_z) - 6371000.0;
293 if (altitude < 0) altitude = 0;
294
295 // Densidad del aire (modelo simplificado)
296 double air_density = 1.225 * exp(-altitude / 8400.0); // kg/m³
297 double dynamic_pressure = 0.5 * air_density * velocity_magnitude * velocity_magnitude;
298 instance->max_dynamic_pressure = std::max(instance->max_dynamic_pressure, dynamic_pressure);
299
300 // SEGUIMIENTO DE MASA
301 if (instance->enable_mass_tracking) {
302 // Calcular masa seca (sin combustible)
303 instance->dry_mass = 0.0;
304 for (const auto& comp : instance->components) {
305 if (!comp.is_fuel_tank) {
306 instance->dry_mass += comp.mass;
307 } else {
308 // Masa del tanque vacío
309 instance->dry_mass += comp.mass * 0.1; // 10% del tanque es estructura
310 }
311 }
312
313 // Masa total = masa seca + combustible actual
314 instance->total_mass = instance->dry_mass + instance->fuel_mass;
315
316 // Calcular centro de masa
317 double total_moment_x = 0.0, total_moment_y = 0.0, total_moment_z = 0.0;
318
319 for (const auto& comp : instance->components) {
320 double comp_mass = comp.mass;
321 if (comp.is_fuel_tank) {
322 // Masa del tanque + combustible
323 comp_mass = comp.mass * 0.1 + instance->fuel_mass * comp.fuel_fraction;
324 }
325
326 total_moment_x += comp_mass * comp.position_x;
327 total_moment_y += comp_mass * comp.position_y;
328 total_moment_z += comp_mass * comp.position_z;
329 }
330
331 if (instance->total_mass > 0) {
332 instance->center_of_mass_x = total_moment_x / instance->total_mass;
333 instance->center_of_mass_y = total_moment_y / instance->total_mass;
334 instance->center_of_mass_z = total_moment_z / instance->total_mass;
335 }
336 }
337
338 // CÁLCULO DE MOMENTOS DE INERCIA
339 if (instance->enable_inertia_calculation) {
340 instance->inertia_xx = 0.0;
341 instance->inertia_yy = 0.0;
342 instance->inertia_zz = 0.0;
343
344 for (const auto& comp : instance->components) {
345 double comp_mass = comp.mass;
346 if (comp.is_fuel_tank) {
347 comp_mass = comp.mass * 0.1 + instance->fuel_mass * comp.fuel_fraction;
348 }
349
350 // Momentos de inercia del componente
351 double Ixx_comp = calculateMomentOfInertia(comp, 'x');
352 double Iyy_comp = calculateMomentOfInertia(comp, 'y');
353 double Izz_comp = calculateMomentOfInertia(comp, 'z');
354
355 // Teorema de ejes paralelos
356 double dx = comp.position_x - instance->center_of_mass_x;
357 double dy = comp.position_y - instance->center_of_mass_y;
358 double dz = comp.position_z - instance->center_of_mass_z;
359
360 instance->inertia_xx += Ixx_comp + comp_mass * (dy*dy + dz*dz);
361 instance->inertia_yy += Iyy_comp + comp_mass * (dx*dx + dz*dz);
362 instance->inertia_zz += Izz_comp + comp_mass * (dx*dx + dy*dy);
363 }
364 }
365
366 // ANÁLISIS ESTRUCTURAL
367 if (instance->enable_structural_analysis) {
368 // Factor de carga estructural
369 double g_force = acceleration / 9.81; // En unidades de g
370 instance->structural_load_factor = g_force;
371
372 // Verificar límites estructurales
373 for (const auto& comp : instance->components) {
374 // Esfuerzo aproximado (simplificado)
375 double cross_sectional_area = M_PI * comp.diameter * comp.thickness;
376 double applied_force = comp.mass * acceleration;
377 double stress = calculateStructuralStress(applied_force, cross_sectional_area);
378
379 // Verificar factor de seguridad
380 double allowable_stress = comp.yield_strength / comp.safety_factor;
381 if (stress > allowable_stress) {
382 // Advertencia: esfuerzo excesivo (en una implementación real,
383 // esto podría disparar un evento de falla estructural)
384 }
385 }
386 }
387
388 // ANÁLISIS DE VIBRACIÓN
389 if (instance->enable_vibration_analysis) {
390 // Calcular frecuencias naturales
391 instance->natural_frequencies.clear();
392 for (const auto& comp : instance->components) {
393 double moment_of_inertia = M_PI * pow(comp.diameter/2.0, 4) / 4.0; // Momento de área
394 double freq = calculateNaturalFrequency(comp.young_modulus, moment_of_inertia,
395 comp.mass, comp.length);
396 instance->natural_frequencies.push_back(freq);
397 }
398 }
399
400 // Este plugin no genera fuerzas directamente, pero modifica las propiedades del vehículo
401 PluginVector3* force_output = data->output_force ? data->output_force : data->force_out;
402 PluginVector3* torque_output = data->output_torque ? data->output_torque : data->torque_out;
403
404 if (force_output) {
405 force_output->x = 0.0;
406 force_output->y = 0.0;
407 force_output->z = 0.0;
408 }
409
410 if (torque_output) {
411 torque_output->x = 0.0;
412 torque_output->y = 0.0;
413 torque_output->z = 0.0;
414 }
415
416 return 0; // Éxito
417}
