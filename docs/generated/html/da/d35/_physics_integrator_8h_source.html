40// Flat vector layout:
41// [0..2] = position (x, y, z)
42// [3..5] = velocity (vx, vy, vz)
43// [6..9] = quaternion orientation (w, x, y, z)
44// [10..12] = angular velocity (wx, wy, wz)
45
46inline Vector3 odePosition(const OdeState& y) {
47 return Vector3(y[0], y[1], y[2]);
48}
49
50inline Vector3 odeVelocity(const OdeState& y) {
51 return Vector3(y[3], y[4], y[5]);
52}
53
54inline Quaternion4 odeOrientation(const OdeState& y) {
55 return Quaternion4(y[6], y[7], y[8], y[9]);
56}
57
58inline Vector3 odeAngularVelocity(const OdeState& y) {
59 return Vector3(y[10], y[11], y[12]);
60}

62inline void odePackVec3(OdeState& y, int offset, const Vector3& v) {
63 y[offset] = v.x();
64 y[offset + 1] = v.y();
65 y[offset + 2] = v.z();
66}
67
68inline void odePackQuat(OdeState& y, const Quaternion4& q) {
69 y[6] = q.w(); y[7] = q.x(); y[8] = q.y(); y[9] = q.z();
70}

72// Build symmetric 3x3 inertia tensor from 6 independent components.
73inline InertiaTensor3 makeInertiaTensor(double ixx, double iyy, double izz,
74 double ixy, double ixz, double iyz) {
75 InertiaTensor3 I;
76 I << ixx, ixy, ixz,
77 ixy, iyy, iyz,
78 ixz, iyz, izz;
79 return I;
80}

82 // Checks whether inertia tensor is diagonal (inertia products ~0).
83inline bool isInertiaDiagonal(const InertiaTensor3& I, double tol = 1e-10) {
84 return std::abs(I(0, 1)) < tol && std::abs(I(0, 2)) < tol && std::abs(I(1, 2)) < tol;
85}
