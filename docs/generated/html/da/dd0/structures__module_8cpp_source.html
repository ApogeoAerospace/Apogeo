20static std::string resolveExistingPath(const std::string& input_path) {
21 namespace fs = std::filesystem;
22 std::error_code ec;
23
24 fs::path original(input_path);
25 if (original.is_absolute() && fs::exists(original, ec)) {
26 return original.string();
27 }
28
29 std::vector<fs::path> candidates;
30 candidates.push_back(original);
31
32 const fs::path cwd = fs::current_path(ec);
33 if (!ec && !cwd.empty()) {
34 candidates.push_back(cwd / original);
35
36 fs::path cursor = cwd;
37 for (int i = 0; i < 6; ++i) {
38 candidates.push_back(cursor / original);
39 if (!cursor.has_parent_path()) {
40 break;
41 }
42 cursor = cursor.parent_path();
43 }
44 }
45
46 for (const auto& candidate : candidates) {
47 if (candidate.empty()) {
48 continue;
49 }
50
51 fs::path normalized = candidate.lexically_normal();
52 if (fs::exists(normalized, ec)) {
53 return normalized.string();
54 }
55 }
56
57 return input_path;
58}

101bool StructuresModule::loadStructuralLimitsFromCsv(const std::string& csv_path) {
102 const std::string resolved_path = resolveExistingPath(csv_path);
103 std::ifstream file(resolved_path);
104 if (!file.is_open()) {
105 return false;
106 }
107
108 std::string line;
109 bool first_line = true;
110 while (std::getline(file, line)) {
111 if (line.empty()) {
112 continue;
113 }
114 if (first_line) {
115 first_line = false;
116 continue;
117 }
118
119 const auto fields = splitCsvLine(line);
120 if (fields.size() < 2) {
121 continue;
122 }
123
124 const std::string& limit_name = fields[0];
125 double value = 0.0;
126 try {
127 value = std::stod(fields[1]);
128 } catch (...) {
129 continue;
130 }
131
132 if (limit_name == "max_g_load") {
133 limits_.max_g_load = value;
134 } else if (limit_name == "warning_g_load") {
135 limits_.warning_g_load = value;
136 } else if (limit_name == "max_dynamic_pressure") {
137 limits_.max_dynamic_pressure = value;
138 } else if (limit_name == "warning_dynamic_pressure") {
139 limits_.warning_dynamic_pressure = value;
140 }
141 }
142
143 return true;
144}
