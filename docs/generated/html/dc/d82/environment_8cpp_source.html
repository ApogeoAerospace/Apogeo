139void calculateAtmosphericProperties(EnvironmentPluginInstance* instance, double altitude) {
140 if (altitude < 0) altitude = 0;
141
142 // Encontrar la capa atmosférica correcta
143 const AtmosphericLayer* layer = nullptr;
144 for (const auto& l : instance->atmosphere_layers) {
145 if (altitude >= l.altitude_base && altitude <= l.altitude_top) {
146 layer = &l;
147 break;
148 }
149 }
150
151 if (!layer) {
152 // Altitud muy alta, usar modelo exponencial simple
153 instance->current_temperature = 200.0; // K
154 instance->current_pressure = STANDARD_PRESSURE * exp(-altitude / 8400.0);
155 instance->current_density = instance->current_pressure / (GAS_CONSTANT_AIR * instance->current_temperature);
156 instance->current_sound_speed = sqrt(GAMMA_AIR * GAS_CONSTANT_AIR * instance->current_temperature);
157 instance->current_viscosity = 1.458e-6 * pow(instance->current_temperature, 1.5) /
158 (instance->current_temperature + 110.4);
159 return;
160 }
161
162 // Calcular temperatura
163 double height_in_layer = altitude - layer->altitude_base;
164 instance->current_temperature = layer->temperature_base - layer->lapse_rate * height_in_layer;
165
166 // Calcular presión
167 if (std::abs(layer->lapse_rate) < 1e-6) {
168 // Capa isotérmica
169 instance->current_pressure = layer->pressure_base *
170 exp(-STANDARD_GRAVITY * height_in_layer / (GAS_CONSTANT_AIR * instance->current_temperature));
171 } else {
172 // Capa con gradiente térmico
173 double temp_ratio = instance->current_temperature / layer->temperature_base;
174 double exponent = STANDARD_GRAVITY / (GAS_CONSTANT_AIR * layer->lapse_rate);
175 instance->current_pressure = layer->pressure_base * pow(temp_ratio, exponent);
176 }
177
178 // Calcular densidad
179 instance->current_density = instance->current_pressure / (GAS_CONSTANT_AIR * instance->current_temperature);
180
181 // Calcular velocidad del sonido
182 instance->current_sound_speed = sqrt(GAMMA_AIR * GAS_CONSTANT_AIR * instance->current_temperature);
183
184 // Calcular viscosidad dinámica (Ley de Sutherland)
185 instance->current_viscosity = 1.458e-6 * pow(instance->current_temperature, 1.5) /
186 (instance->current_temperature + 110.4);
187}

189double calculateLocalGravity(double altitude) {
190 double r = EARTH_RADIUS + altitude;
191 if (r < EARTH_RADIUS * 0.5) { // Protección contra valores incorrectos
192 return STANDARD_GRAVITY;
193 }
194 return GRAVITATIONAL_CONSTANT * EARTH_MASS / (r * r);
195}

197void calculateWindEffects(EnvironmentPluginInstance* instance, double altitude, double time) {
198 // Modelo de viento simplificado con variación altitudinal
199 double altitude_km = altitude / 1000.0;
200
201 // Viento base (jet stream aproximado)
202 double base_wind_speed = 0.0;
203 if (altitude_km > 8.0 && altitude_km < 15.0) {
204 // Jet stream entre 8-15 km
205 base_wind_speed = 50.0 * sin(M_PI * (altitude_km - 8.0) / 7.0);
206 } else if (altitude_km > 15.0) {
207 // Viento estratosférico
208 base_wind_speed = 20.0 * exp(-(altitude_km - 15.0) / 10.0);
209 } else {
210 // Viento troposférico
211 base_wind_speed = 10.0 * (altitude_km / 8.0);
212 }
213
214 // Variación temporal (simulación de cambios meteorológicos)
215 double temporal_variation = 0.3 * sin(time / 3600.0) + 0.2 * sin(time / 1800.0);
216
217 instance->wind_magnitude = base_wind_speed * (1.0 + temporal_variation);
218 instance->wind_direction += 0.001 * sin(time / 900.0); // Rotación lenta
219
220 // Componentes del viento
221 instance->wind_speed_x = instance->wind_magnitude * cos(instance->wind_direction);
222 instance->wind_speed_y = instance->wind_magnitude * sin(instance->wind_direction);
223 instance->wind_speed_z = 0.1 * instance->wind_magnitude * sin(time / 600.0); // Viento vertical
224
225 // Turbulencia
226 if (instance->enable_turbulence) {
227 double turb_intensity = instance->turbulence_intensity;
228
229 // Aumentar turbulencia cerca del suelo y en jet stream
230 if (altitude_km < 1.0) {
231 turb_intensity *= (2.0 - altitude_km); // Turbulencia de superficie
232 } else if (altitude_km > 8.0 && altitude_km < 15.0) {
233 turb_intensity *= 1.5; // Turbulencia del jet stream
234 }
235
236 // Generar componentes turbulentas
237 double turb_x = instance->turbulence_distribution(instance->random_generator) * turb_intensity;
238 double turb_y = instance->turbulence_distribution(instance->random_generator) * turb_intensity;
239 double turb_z = instance->turbulence_distribution(instance->random_generator) * turb_intensity * 0.5;
240
241 instance->wind_speed_x += turb_x;
242 instance->wind_speed_y += turb_y;
243 instance->wind_speed_z += turb_z;
244 }

308PLUGIN_EXPORT int32_t plugin_tick(PluginHandle handle, PluginTickData* data) {
309 if (!handle || !data || !data->state_buffer) {
310 return -1; // Error: parámetros inválidos
311 }
312
313 // Compatibilidad con ambas versiones de la API
314 PluginVector3* force_output = data->output_force ? data->output_force : data->force_out;
315 PluginVector3* torque_output = data->output_torque ? data->output_torque : data->torque_out;
316
317 if (!force_output) {
318 return -1; // Error: sin puntero de salida para fuerzas
319 }
320
321 EnvironmentPluginInstance* instance = reinterpret_cast<EnvironmentPluginInstance*>(handle);
322 if (!instance->initialized) {
323 return -2; // Error: plugin no inicializado
324 }
325
326 // Actualizar tiempo de simulación
327 instance->simulation_time += data->delta_time;
328
329 // Obtener estado actual del FlatBuffer
330 auto state = state_vector::GetGeneralState(data->state_buffer);
331 if (!state) return -3;
332
333 // Extraer datos del estado
334 double pos_x = state->position()->x();
335 double pos_y = state->position()->y();
336 double pos_z = state->position()->z();
337 double vel_x = state->velocity()->x();
338 double vel_y = state->velocity()->y();
339 double vel_z = state->velocity()->z();
340
341 // Calcular altitud
342 double altitude = sqrt(pos_x*pos_x + pos_y*pos_y + pos_z*pos_z) - EARTH_RADIUS;
343 if (altitude < 0) altitude = 0;
344
345 // MODELO ATMOSFÉRICO
346 if (instance->enable_atmospheric_model) {
347 calculateAtmosphericProperties(instance, altitude);
348 }
349
350 // MODELO GRAVITACIONAL
351 if (instance->enable_gravity_variation) {
352 instance->local_gravity = calculateLocalGravity(altitude);
353 instance->gravity_gradient = -2.0 * instance->local_gravity / (EARTH_RADIUS + altitude);
354 }
355
356 // MODELO DE VIENTO
357 if (instance->enable_wind_effects) {
358 calculateWindEffects(instance, altitude, instance->simulation_time);
359 }
360
361 // Inicializar fuerzas ambientales
362 double force_x = 0.0, force_y = 0.0, force_z = 0.0;
363
364 // EFECTOS DE VIENTO (como fuerza de arrastre diferencial)
365 if (instance->enable_wind_effects) {
366 // Velocidad relativa del viento
367 double relative_wind_x = instance->wind_speed_x - vel_x;
368 double relative_wind_y = instance->wind_speed_y - vel_y;
369 double relative_wind_z = instance->wind_speed_z - vel_z;
370
371 double relative_wind_magnitude = sqrt(relative_wind_x*relative_wind_x +
372 relative_wind_y*relative_wind_y +
373 relative_wind_z*relative_wind_z);
374
375 if (relative_wind_magnitude > 0.1) {
376 // Fuerza de viento proporcional a la velocidad relativa al cuadrado
377 double wind_drag_coefficient = 0.1; // Coeficiente simplificado
378 double reference_area = 1.0; // m² (área de referencia)
379
380 double wind_force_magnitude = 0.5 * instance->current_density *
381 relative_wind_magnitude * relative_wind_magnitude *
382 wind_drag_coefficient * reference_area;
383
384 // Dirección de la fuerza (en dirección del viento relativo)
385 force_x += wind_force_magnitude * (relative_wind_x / relative_wind_magnitude);
386 force_y += wind_force_magnitude * (relative_wind_y / relative_wind_magnitude);
387 force_z += wind_force_magnitude * (relative_wind_z / relative_wind_magnitude);
388 }
389 }
390
391 // EFECTOS DE CORIOLIS (si están habilitados)
392 if (instance->enable_coriolis_effect) {
393 // Fuerza de Coriolis: F = -2m * Ω × v
394 double omega = EARTH_ROTATION_RATE;
395 double mass = 1000.0; // Masa estimada (debería venir del plugin de estructuras)
396
397 // Componentes de la fuerza de Coriolis (simplificado)
398 double coriolis_x = -2.0 * mass * omega * vel_y * sin(instance->latitude);
399 double coriolis_y = 2.0 * mass * omega * vel_x * sin(instance->latitude);
400 double coriolis_z = 0.0; // Componente vertical simplificada
401
402 force_x += coriolis_x;
403 force_y += coriolis_y;
404 force_z += coriolis_z;
405 }
406
407 // VARIACIÓN GRAVITACIONAL
408 if (instance->enable_gravity_variation) {
409 // CORRECCIÓN: Usar gravedad completa, no diferencia
410 double mass = 1000.0; // TODO: obtener masa real del state
411
412 // Calcular distancia al centro de la Tierra
413 double distance_to_center = sqrt(pos_x*pos_x + pos_y*pos_y + pos_z*pos_z);
414
415 // Si estamos muy cerca del centro (error), usar posición por defecto
416 if (distance_to_center < EARTH_RADIUS * 0.5) {
417 // Asumir superficie de la Tierra
418 distance_to_center = EARTH_RADIUS;
419 pos_z = EARTH_RADIUS; // Colocar en superficie
420 }
421
422 // Calcular magnitud de gravedad con ley del cuadrado inverso
423 double gravity_magnitude = GRAVITATIONAL_CONSTANT * EARTH_MASS /
424 (distance_to_center * distance_to_center);
425
426 // DIRECCIÓN: Hacia el centro de la Tierra
427 double unit_x = -pos_x / distance_to_center;
428 double unit_y = -pos_y / distance_to_center;
429 double unit_z = -pos_z / distance_to_center;
430
431 // Fuerza gravitacional F = mg
432 force_x += mass * gravity_magnitude * unit_x;
433 force_y += mass * gravity_magnitude * unit_y;
434 force_z += mass * gravity_magnitude * unit_z;
435
436 // Perturbación J2 (achatamiento terrestre) - importante para órbitas
437 if (distance_to_center > EARTH_RADIUS) {
438 const double J2 = 1.08262668e-3;
439 double r_ratio = EARTH_RADIUS / distance_to_center;
440 double z_over_r = pos_z / distance_to_center;
441
442 double j2_factor = 1.5 * J2 * r_ratio * r_ratio;
443 double j2_radial = j2_factor * (1.0 - 5.0 * z_over_r * z_over_r);
444 double j2_axial = j2_factor * (3.0 - 5.0 * z_over_r * z_over_r);
445
446 // Aplicar correcciones J2
447 force_x *= (1.0 + j2_radial);
448 force_y *= (1.0 + j2_radial);
449 force_z += mass * gravity_magnitude * j2_axial * z_over_r;
450 }
451 }
452
453 // Asignar fuerzas de salida
454 force_output->x = static_cast<float>(force_x);
455 force_output->y = static_cast<float>(force_y);
456 force_output->z = static_cast<float>(force_z);
457
458 // No hay torques ambientales significativos en este modelo simplificado
459 if (torque_output) {
460 torque_output->x = 0.0;
461 torque_output->y = 0.0;
462 torque_output->z = 0.0;
463 }
464
465 return 0; // Éxito
466}
