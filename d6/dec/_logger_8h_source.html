120 /**
121 * @brief Emits a formatted message by level and component.
122 * @param level Message severity level.
123 * @param message Message text.
124 * @param component Source component name.
125 */
126 void log(LogLevel level, const std::string& message, const std::string& component = "") {
127 if (level < current_level_) {
128 return;
129 }
130
131 std::function<void(LogLevel, const std::string&, const std::string&)> structured_sink;
132
133 {
134 std::lock_guard<std::mutex> lock(mutex_);
135
136 auto now = std::chrono::system_clock::now();
137 auto time_t = std::chrono::system_clock::to_time_t(now);
138 auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(
139 now.time_since_epoch()) % 1000;
140
141 std::stringstream ss;
142 ss << std::put_time(std::localtime(&time_t), "%Y-%m-%d %H:%M:%S");
143 ss << '.' << std::setfill('0') << std::setw(3) << ms.count();
144 ss << " [" << levelToString(level) << "]";
145
146 if (!component.empty()) {
147 ss << " [" << component << "]";
148 }
149
150 ss << " " << message;
151
152 std::string log_line = ss.str();
153
154 recent_logs_.push_back(LogEntry{level, message, component});
155 if (recent_logs_.size() > max_buffered_logs_) {
156 recent_logs_.pop_front();
157 }
158
159 // Output to console
160 if (console_output_enabled_) {
161 std::cout << log_line << std::endl;
162 }
163
164 // Output to file if available
165 if (!log_file_path_.empty()) {
166 if (!log_file_.is_open()) {
167 log_file_.open(log_file_path_, std::ios::app);
168 }
169 if (log_file_.is_open()) {
170 log_file_ << log_line << std::endl;
171 log_file_.flush();
172 log_file_.close();
173 }
174 }
175
176 structured_sink = structured_sink_;
177 }
178
179 if (structured_sink) {
180 try {
181 structured_sink(level, message, component);
182 } catch (...) {
183 // Structured sink does not break normal logging path.
184 }
185 }
186 }
