297SimulationDataPoint OutputManager::extractDataPoint(const state_vector::GeneralState* state, double time, double utc_time, int tick) {
298 SimulationDataPoint point;
299 point.time = time;
300 point.utc_time = utc_time;
301
302 // Position
303 if (state->position()) {
304 point.position_x = state->position()->x();
305 point.position_y = state->position()->y();
306 point.position_z = state->position()->z();
307 }
308
309 // Velocity
310 if (state->velocity()) {
311 point.velocity_x = state->velocity()->x();
312 point.velocity_y = state->velocity()->y();
313 point.velocity_z = state->velocity()->z();
314 }
315
316 // Orientation (quaternion)
317 if (state->orientation()) {
318 point.orientation_x = state->orientation()->x();
319 point.orientation_y = state->orientation()->y();
320 point.orientation_z = state->orientation()->z();
321 point.orientation_w = state->orientation()->w();
322 }
323
324 // Atmospheric data
325 point.atm_density = state->atm_density();
326 point.atm_pressure = state->atm_pressure();
327 point.atm_temperature = state->atm_temperature();
328
329 // Gravity
330 if (state->gravity()) {
331 point.gravity_x = state->gravity()->x();
332 point.gravity_y = state->gravity()->y();
333 point.gravity_z = state->gravity()->z();
334 } else {
335 point.gravity_x = 0.0;
336 point.gravity_y = 0.0;
337 point.gravity_z = 0.0;
338 }
339
340 // Wind
341 if (state->wind_velocity()) {
342 point.wind_speed_x = state->wind_velocity()->x();
343 point.wind_speed_y = state->wind_velocity()->y();
344 point.wind_speed_z = state->wind_velocity()->z();
345 } else {
346 point.wind_speed_x = 0.0;
347 point.wind_speed_y = 0.0;
348 point.wind_speed_z = 0.0;
349 }
350
351 // Angular velocity
352 if (state->angular_velocity()) {
353 point.angular_velocity_x = state->angular_velocity()->x();
354 point.angular_velocity_y = state->angular_velocity()->y();
355 point.angular_velocity_z = state->angular_velocity()->z();
356 } else {
357 point.angular_velocity_x = 0.0;
358 point.angular_velocity_y = 0.0;
359 point.angular_velocity_z = 0.0;
360 }
361
362 // Mass and properties
363 point.total_mass = state->total_mass();
364 if (state->cg_location()) {
365 point.cg_x = state->cg_location()->x();
366 point.cg_y = state->cg_location()->y();
367 point.cg_z = state->cg_location()->z();
368 } else {
369 point.cg_x = 0.0;
370 point.cg_y = 0.0;
371 point.cg_z = 0.0;
372 }
373
374 // Aerodynamic data
375 point.mach_number = state->mach_number();
376 point.dynamic_pressure = state->dynamic_pressure();
377 point.angle_of_attack = state->angle_of_attack();
378 point.sideslip_angle = state->sideslip_angle();
379
380 return point;
381}

486void OutputManager::writerLoop() {
487 for (;;) {
488 PendingPoint pending;
489 std::function<void(const SimulationDataPoint&, int)> telemetry_callback;
490 {
491 std::unique_lock<std::mutex> lock(queue_mutex_);
492 queue_cv_.wait(lock, [this]() { return !writer_running_.load() || !pending_points_.empty(); });
493
494 if (!writer_running_.load() && pending_points_.empty()) {
495 break;
496 }
497
498 pending = std::move(pending_points_.front());
499 pending_points_.pop_front();
500 }
501
502 {
503 std::lock_guard<std::mutex> lock(data_mutex_);
504
505 if (!initialized_) {
506 continue;
507 }
508
509 data_points_.push_back(pending.point);
510
511 if (output_csv_ && csv_file_ && csv_file_->is_open()) {
512 writeDataPointCSV(pending.point, pending.tick);
513 }
514
515 if (output_json_ && json_file_ && json_file_->is_open()) {
516 writeDataPointJSON(pending.point, pending.tick);
517 }
518
519 if (output_binary_ && binary_file_ && binary_file_->is_open()) {
520 binary_file_->write(reinterpret_cast<const char*>(&pending.point), sizeof(SimulationDataPoint));
521 }
522
523 last_recorded_tick_ = pending.tick;
524 telemetry_callback = realtime_telemetry_callback_;
525 }
526
527 if (telemetry_callback) {
528 try {
529 telemetry_callback(pending.point, pending.tick);
530 } catch (...) {
531 // Telemetry callback must not break async output writer.
532 }
533 }
534 }
